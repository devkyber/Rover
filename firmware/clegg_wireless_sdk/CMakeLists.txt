cmake_minimum_required(VERSION 3.13)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(clegg_wireless_sdk C CXX ASM)
pico_sdk_init()
add_executable(clegg_wireless_sdk clegg_wireless_sdk.c nina.c)
target_include_directories(clegg_wireless_sdk PRIVATE ${CLEGG_GENERATED_DIR})
target_link_libraries(clegg_wireless_sdk pico_stdlib pico_multicore pico_rand hardware_spi)
pico_enable_stdio_uart(clegg_wireless_sdk 0)
pico_enable_stdio_usb(clegg_wireless_sdk 1)
target_compile_definitions(clegg_wireless_sdk PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=100000)
pico_add_extra_outputs(clegg_wireless_sdk)
